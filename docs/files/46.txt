function parseJsonc(text) {
    const stripped = text.replace(/\/\/.*$/gm, '').replace(/\/\*[\s\S]*?\*\//g, '');
    return JSON.parse(stripped);

let cachedHeaderEnds = null;        // Map<headerLine, endLine>
let cachedHiddenRanges = null;      // all collapse-char ranges (no cursor filter)
let cachedCurlyRanges = null;       // all curly-quote ranges (no cursor filter)

function invalidateCache() {
    cacheDocVersion = -1;
    cacheDocUri = null;
    cachedHeaders = null;
    cachedHeaderEnds = null;
    cachedHiddenRanges = null;
    cachedCurlyRanges = null;
    cachedReadTimeRanges = null;
    cachedNumberRanges = null;
}

function ensureCache(document) {
    const uri = document.uri.toString();
    if (cacheDocVersion === document.version && cacheDocUri === uri) return;
    cacheDocVersion = document.version;
    cacheDocUri = uri;
    cachedHeaders = null;
    cachedHeaderEnds = null;
    cachedHiddenRanges = null;
    cachedCurlyRanges = null;
    cachedReadTimeRanges = null;
    cachedNumberRanges = null;
}

function getHeaders(document) {
    ensureCache(document);
    if (!cachedHeaders) cachedHeaders = parseHeaders(document);
    return cachedHeaders;
}

function getHeaderEnd(document, headerLine) {
    ensureCache(document);
    if (!cachedHeaderEnds) {
        cachedHeaderEnds = new Map();
        for (const h of getHeaders(document)) {
            cachedHeaderEnds.set(h.line, findHeaderEnd(document, h.line));
        }
    }
    return cachedHeaderEnds.get(headerLine);
}

const READ_TIME_RE = /\s*\{[^}]*\}\s*$/;
const HEADER_RE = /^(\t*)(.+?)\s*\[$/;

function findThemeFile() {
    const themeId = vscode.workspace.getConfiguration('workbench').get('colorTheme');
    if (!themeId) return null;

    for (const ext of vscode.extensions.all) {
        const pkgPath = path.join(ext.extensionPath, 'package.json');
        if (!fs.existsSync(pkgPath)) continue;

        let pkg;
        try { pkg = parseJsonc(fs.readFileSync(pkgPath, 'utf8')); }
        catch { continue; }

        const themes = pkg.contributes?.themes;
        if (!Array.isArray(themes)) continue;

        for (const t of themes) {
            if (t.id === themeId || t.label === themeId) {
                const themePath = path.resolve(ext.extensionPath, t.path);
                if (fs.existsSync(themePath)) return themePath;
            }
        }
    }
    return null;
}

function loadThemeTokenColors(themePath) {
    if (!themePath || !fs.existsSync(themePath)) return [];

    let theme;
    try {
        const content = fs.readFileSync(themePath, 'utf8');
        theme = parseJsonc(content);
    }
    catch { return []; }

    let tokenColors = theme.tokenColors || [];

    if (theme.include) {
        const parentPath = path.resolve(path.dirname(themePath), theme.include);
        const parentColors = loadThemeTokenColors(parentPath);
        tokenColors = [...parentColors, ...tokenColors];
    }

    return tokenColors;
}

function getColorForScope(targetScope) {
    const themePath = findThemeFile();
    if (!themePath) return null;

    const tokenColors = loadThemeTokenColors(themePath);
    let bestColor = null;

    for (const rule of tokenColors) {
        if (!rule.scope || !rule.settings?.foreground) continue;

        const scopes = Array.isArray(rule.scope)
            ? rule.scope
            : rule.scope.split(/\s*,\s*/);

        for (const scope of scopes) {
            if (targetScope === scope || targetScope.startsWith(scope + '.')) {
                bestColor = rule.settings.foreground;
            }
        }
    }

    const customColors = vscode.workspace
        .getConfiguration('editor')
        .get('tokenColorCustomizations');

    if (customColors?.textMateRules) {
        for (const rule of customColors.textMateRules) {
            if (!rule.scope || !rule.settings?.foreground) continue;
            const scopes = Array.isArray(rule.scope)
                ? rule.scope
                : rule.scope.split(/\s*,\s*/);
            for (const scope of scopes) {
                if (targetScope === scope || targetScope.startsWith(scope + '.')) {
                    bestColor = rule.settings.foreground;
                }
            }
        }
    }

    return bestColor;
}

// Curly quote helpers
// ─────────────────────────────────────────────

function distanceToWhitespace(str, pos, direction) {
    let distance = 0;
    let i = pos + direction;
    while (i >= 0 && i < str.length && !/\s/.test(str[i])) {
        distance++;
        i += direction;
    }
    if (i < 0 || i >= str.length) distance++;
    return distance;
}

function curlyForDouble(text, index) {
    const leftDist = distanceToWhitespace(text, index, -1);
    const rightDist = distanceToWhitespace(text, index, 1);
    if (leftDist < rightDist) return '\u201C';
    if (rightDist < leftDist) return '\u201D';
    return '\u201C';
}

function curlyForSingle(text, index) {
    const leftChar = index > 0 ? text[index - 1] : ' ';
    const rightChar = index < text.length - 1 ? text[index + 1] : ' ';
    if (/[a-zA-Z]/.test(leftChar) && /[a-zA-Z]/.test(rightChar)) {
        return '\u2019';
    }
    const leftDist = distanceToWhitespace(text, index, -1);
    const rightDist = distanceToWhitespace(text, index, 1);
    if (leftDist < rightDist) return '\u2018';
    if (rightDist < leftDist) return '\u2019';
    return '\u2018';
}

// ─────────────────────────────────────────────
// Header parsing (bracket-based)
// ─────────────────────────────────────────────

function parseHeaders(document) {
    const headers = [];
    for (let i = 0; i < document.lineCount; i++) {
        const text = document.lineAt(i).text;
        const match = text.match(HEADER_RE);
        if (match) {
            const level = match[1].length + 1;
            const title = match[2].replace(READ_TIME_RE, '').trim();
            headers.push({ line: i, level, title });
        }
    }
    return headers;
}

function findHeaderEnd(document, headerLine) {
    let depth = 1;
    for (let i = headerLine + 1; i < document.lineCount; i++) {
        const line = document.lineAt(i).text;
        for (const ch of line) {
            if (ch === '[') depth++;
            if (ch === ']') depth--;
            if (depth === 0) return i;
        }
    }
    return document.lineCount - 1;
}

// ─────────────────────────────────────────────
// Read time
// ─────────────────────────────────────────────

function formatTime(minutes) {
    const h = Math.floor(minutes / 60);
    const m = Math.round(minutes % 60);
    if (h > 0) return `${h}h ${m}m`;
    return `${m}m`;
}

function countWordsInRange(document, startLine, endLine) {
    let count = 0;
    for (let i = startLine; i < endLine; i++) {
        let lineText = document.lineAt(i).text;
        lineText = lineText.replace(HEADER_RE, '').replace(READ_TIME_RE, '');
        const trimmed = lineText.trim();
        if (trimmed.length > 0) {
            count += trimmed.split(/\s+/).length;
        }
    }
    return count;
}

function buildNumberDecorations(document) {
    const headers = getHeaders(document);
    if (headers.length === 0) return [];

    const numberRanges = [];

    // Assign hierarchical numbers to headers by sibling position
    const headerNumbers = new Map();
    const counters = [0]; // stack tracking sibling count per depth

    for (const header of headers) {
        while (counters.length > header.level) counters.pop();
        while (counters.length < header.level) counters.push(0);
        counters[counters.length - 1]++;
        headerNumbers.set(header.line, counters.join('.'));
    }

    // Decorate each header and its direct content lines
    for (const header of headers) {
        const endLine = getHeaderEnd(document, header.line);
        const headerNum = headerNumbers.get(header.line);

        // Header line decoration
        numberRanges.push({
            range: new vscode.Range(header.line, 0, header.line, 0),
            renderOptions: {
                before: {
                    contentText: `${headerNum} `,
                    color: '#888888',
                }
            }
        });

        // Collect ranges of all nested child headers to exclude
        const childRanges = [];
        for (const h of headers) {
            if (h.line > header.line && h.line <= endLine) {
                childRanges.push({
                    start: h.line,
                    end: getHeaderEnd(document, h.line)
                });
            }
        }

        // Number direct content lines sequentially
        let contentCounter = 0;
        for (let i = header.line + 1; i <= endLine; i++) {
            // Skip lines inside nested headers
            let inChild = false;
            for (const cr of childRanges) {
                if (i >= cr.start && i <= cr.end) { inChild = true; break; }
            }
            if (inChild) continue;

            const text = document.lineAt(i).text;
            const trimmed = text.trim();
            if (!trimmed || trimmed === ']') continue;

            contentCounter++;
            numberRanges.push({
                range: new vscode.Range(i, 0, i, 0),
                renderOptions: {
                    before: {
                        contentText: `${headerNum}.${contentCounter} `,
                        color: '#888888',
                    }
                }
            });
        }
    }

    return numberRanges;
}

// ─────────────────────────────────────────────
// Content decoration cache builders
// ─────────────────────────────────────────────

function buildContentDecorations(document, config) {
    ensureCache(document);

    const hiddenRanges = [];
    const curlyRanges = [];

    for (let i = 0; i < document.lineCount; i++) {
        const text = document.lineAt(i).text;

        // Collapse chars and curly quotes — store ALL positions
        for (let col = 0; col < text.length; col++) {
            const ch = text[col];

            if (config.collapseChars.includes(ch)) {
                hiddenRanges.push({
                    range: new vscode.Range(i, col, i, col + 1),
                    _line: i,
                    _col: col,
                });
            }

            if (config.curlyQuotes) {
                let replacement = null;
                if (ch === '"') replacement = curlyForDouble(text, col);
                else if (ch === "'") replacement = curlyForSingle(text, col);

                if (replacement) {
                    curlyRanges.push({
                        range: new vscode.Range(i, col, i, col + 1),
                        renderOptions: {
                            after: {
                                contentText: replacement,
                                color: dialogColor
                            }
                        },
                        _line: i,
                        _col: col,
                    });
                }
            }
        }
    }

    return { hiddenRanges, curlyRanges };
}

function buildReadTimeDecorations(document, config) {
    const readTimeRanges = [];
    if (!config.readTime) return readTimeRanges;

    const headers = getHeaders(document);
    for (const header of headers) {
        const endLine = getHeaderEnd(document, header.line);
        const words = countWordsInRange(document, header.line + 1, endLine);
        const minutes = words / config.readTimeWpm;
        const timeStr = minutes >= 0.5 ? formatTime(minutes) : '<1m';

        const lineText = document.lineAt(header.line).text;
        const bracketPos = lineText.lastIndexOf('[');

        if (bracketPos > 0) {
            readTimeRanges.push({
                range: new vscode.Range(header.line, bracketPos, header.line, bracketPos),
                renderOptions: {
                    before: {
                        contentText: `${timeStr} `,
                        color: '#888888'
                    }
                }
            });
        }
    }

    return readTimeRanges;
}

// ─────────────────────────────────────────────
// Cached content decoration accessors
// ─────────────────────────────────────────────

function ensureContentDecorations(document, config) {
    ensureCache(document);
    if (!cachedHiddenRanges) {
        const result = buildContentDecorations(document, config);
        cachedHiddenRanges = result.hiddenRanges;
        cachedCurlyRanges = result.curlyRanges;
    }
    if (!cachedReadTimeRanges) {
        cachedReadTimeRanges = buildReadTimeDecorations(document, config);
    }
    if (!cachedNumberRanges) {
        cachedNumberRanges = config.lineNumbers
            ? buildNumberDecorations(document)
            : [];
    }
}

// ─────────────────────────────────────────────
// Cursor-aware filtering
// ─────────────────────────────────────────────

function isCursorAdjacent(line, col, cursorLine, cursorChar) {
    return line === cursorLine && cursorChar >= col && cursorChar <= col + 1;
}

function filterCursorRanges(ranges, cursorLine, cursorChar) {
    const result = [];
    for (let i = 0; i < ranges.length; i++) {
        const r = ranges[i];
        if (!isCursorAdjacent(r._line, r._col, cursorLine, cursorChar)) {
            result.push(r);
        }
    }
    return result;
}

// ─────────────────────────────────────────────
// Paragraph indent helpers
// ─────────────────────────────────────────────

const INDENT_RE = new RegExp('^(\t*)' + INDENT_STR);
const LIST_RE = /^[-•]|\d+\./;

function lineNeedsIndent(text, config) {
    const afterTabs = text.replace(/^\t*/, '');
    const trimmed = afterTabs.replace(new RegExp('^' + INDENT_STR), '').trimStart();
    if (!trimmed || trimmed === ']') return false;
    if (HEADER_RE.test(text.replace(INDENT_STR, ''))) return false;
    if (LIST_RE.test(trimmed)) return false;
    if (config.lineNumbers) return false;
    return true;
}

function lineHasIndent(text) {
    return INDENT_RE.test(text);
}

async function stripAllIndents(editor) {
    const document = editor.document;
    const edit = new vscode.WorkspaceEdit();
    let hasChanges = false;

    for (let i = 0; i < document.lineCount; i++) {
        const text = document.lineAt(i).text;
        const match = text.match(INDENT_RE);
        if (match) {
            const tabLen = match[1].length;
            edit.delete(document.uri, new vscode.Range(i, tabLen, i, tabLen + INDENT_STR.length));
            hasChanges = true;
        }
    }

    if (hasChanges) {
        isApplyingEdits = true;
        await vscode.workspace.applyEdit(edit);
        isApplyingEdits = false;
    }
}

function applyDecorations(editor) {
    if (!editor || !enabled) {
        if (editor) {
            editor.setDecorations(hiddenDecorationType, []);
            editor.setDecorations(curlyQuoteDecorationType, []);
            editor.setDecorations(readTimeDecorationType, []);
            editor.setDecorations(lineNumberDecorationType, []);
        }
        return;
    }

    const config = getConfig();

    if (!isActiveFor(editor.document)) {
        editor.setDecorations(hiddenDecorationType, []);
        editor.setDecorations(curlyQuoteDecorationType, []);
        editor.setDecorations(readTimeDecorationType, []);
        editor.setDecorations(lineNumberDecorationType, []);
        return;
    }

    const document = editor.document;
    const cursorLine = editor.selection.active.line;
    const cursorChar = editor.selection.active.character;

    // Build/reuse cached content decorations
    ensureContentDecorations(document, config);

    // Apply read time + line numbers as-is (not cursor-dependent)
    editor.setDecorations(readTimeDecorationType, cachedReadTimeRanges);
    editor.setDecorations(lineNumberDecorationType, cachedNumberRanges);

    // Filter collapse chars and curly quotes near cursor
    const filteredHidden = filterCursorRanges(cachedHiddenRanges, cursorLine, cursorChar);
    const filteredCurly = filterCursorRanges(cachedCurlyRanges, cursorLine, cursorChar);

    editor.setDecorations(hiddenDecorationType, filteredHidden);
    editor.setDecorations(curlyQuoteDecorationType, filteredCurly);
}

function updateStatusBar(editor) {
    if (!editor || !enabled) {
        statusBarItem.hide();
        return;
    }

    const config = getConfig();
    if (!config.readTime || !isActiveFor(editor.document)) {
        statusBarItem.hide();
        return;
    }

    const document = editor.document;
    const headers = getHeaders(document);

    if (headers.length === 0) {
        statusBarItem.hide();
        return;
    }

    const totalWords = countWordsInRange(document, 0, document.lineCount);
    const totalMin = totalWords / config.readTimeWpm;
    const totalStr = totalMin >= 0.5 ? formatTime(totalMin) : '<1m';

    const cursorLine = editor.selection.active.line;
    const wordsBefore = countWordsInRange(document, 0, cursorLine);
    const cursorMin = wordsBefore / config.readTimeWpm;
    const cursorStr = cursorMin >= 0.5 ? formatTime(cursorMin) : '<1m';

    statusBarItem.text = `🕮 ${cursorStr} / ${totalStr}`;
    statusBarItem.show();
}

// ─────────────────────────────────────────────
// Uppercase + indent transforms
// ─────────────────────────────────────────────

async function applyTransforms(editor) {
    if (!editor || !enabled || isApplyingEdits) return;

    const config = getConfig();
    if (!isActiveFor(editor.document)) return;

    const document = editor.document;
    const edit = new vscode.WorkspaceEdit();
    let hasChanges = false;

    for (let i = 0; i < document.lineCount; i++) {
        const line = document.lineAt(i);
        const text = line.text;

        // Uppercase headers
        if (config.uppercaseHeaders) {
            const match = text.match(/^(\t*)(.+?)\s*\[$/);
            if (match) {
                const indent = match[1];
                const title = match[2];
                const upper = title.toUpperCase();
                if (title !== upper) {
                    const newText = `${indent}${upper} [`;
                    edit.replace(document.uri, line.range, newText);
                    hasChanges = true;
                    continue; // header line, skip indent logic
                }
            }
        }

        // Paragraph indent: insert or remove em quads
        const needs = lineNeedsIndent(text, config);
        const has = lineHasIndent(text);

        if (needs && !has) {
            const tabLen = text.match(/^\t*/)[0].length;
            edit.insert(document.uri, new vscode.Position(i, tabLen), INDENT_STR);
            hasChanges = true;
        } else if (!needs && has) {
            const match = text.match(INDENT_RE);
            if (match) {
                const tabLen = match[1].length;
                edit.delete(document.uri, new vscode.Range(i, tabLen, i, tabLen + INDENT_STR.length));
                hasChanges = true;
            }
        }
    }

    if (hasChanges) {
        isApplyingEdits = true;
        await vscode.workspace.applyEdit(edit);
        isApplyingEdits = false;
    }
}

function formatJav(text) {
    const lines = text.split('\n').map(l => l.replace(/\r$/, ''));
    const result = [];
    let currentIndent = 0;
    const maxIndent = 5;

    const listPattern = /^[-•]|\d+\./;

  for (let i = 0; i < lines.length; i++) {
    let line = lines[i].replace(/^[\t ]+/, '').replace(/[\t ]+$/, '');
    if (line === '') continue;

    if (line === ']') {
      currentIndent = Math.max(0, currentIndent - 1);
      result.push('\t'.repeat(currentIndent) + ']');
      continue;
    }

    const headerMatch = line.match(/^(.+?)\s*(\{[^}]*\})?\s*\[$/);
    if (headerMatch) {
      const title = headerMatch[1].trim();
      result.push('\t'.repeat(currentIndent) + title + ' [');
      currentIndent = Math.min(currentIndent + 1, maxIndent);
      continue;
    }

    const isListItem = listPattern.test(line);
    if (isListItem) {
      const originalLine = lines[i];
      const leadingTabs = originalLine.match(/^\t*/)[0].length;
      const strippedContent = originalLine.replace(/^[\t ]+/, '');
      const indent = Math.max(leadingTabs, currentIndent);
      result.push('\t'.repeat(indent) + strippedContent);
      continue;
    }

    line = line.replace(/[ \t]+/g, ' ');
    result.push('\t'.repeat(currentIndent) + line);
    }

    return result.join('\n');
}

const symbolProvider = {
    provideDocumentSymbols(document) {
        const symbols = [];
        const headers = getHeaders(document);
        const nodeStack = [];

        for (let i = 0; i < headers.length; i++) {
            const header = headers[i];
            const endLine = getHeaderEnd(document, header.line);

            const fullRange = new vscode.Range(
                header.line, 0,
                endLine, document.lineAt(endLine).text.length
            );
            const selectionRange = document.lineAt(header.line).range;

            const numeral = ['🟢', '🟡', '🔴', '🩷', '🟣', '🔵'][header.level - 1] || '🟢';
            const symbol = new vscode.DocumentSymbol(
                `${numeral} ${header.title}`,
                '',
                vscode.SymbolKind.File,
                fullRange,
                selectionRange
            );

            while (nodeStack.length > 0 && nodeStack[nodeStack.length - 1].level >= header.level) {
                nodeStack.pop();
            }

            if (nodeStack.length > 0) {
                nodeStack[nodeStack.length - 1].symbol.children.push(symbol);
            } else {
                symbols.push(symbol);
            }

            nodeStack.push({ level: header.level, symbol });
        }

        return symbols;
    }
};

const foldingProvider = {
    provideFoldingRanges(document) {
        const ranges = [];
        const headers = getHeaders(document);

        for (const header of headers) {
            let endLine = getHeaderEnd(document, header.line);

            while (endLine > header.line && document.lineAt(endLine).isEmptyOrWhitespace) {
                endLine--;
            }

            if (endLine > header.line) {
                ranges.push(new vscode.FoldingRange(
                    header.line,
                    endLine,
                    vscode.FoldingRangeKind.Region
                ));
            }
        }

        return ranges;
    }
};

function activate(context) {
    hiddenDecorationType = vscode.window.createTextEditorDecorationType({
        letterSpacing: '-1em',
        opacity: '0',
        textDecoration: 'none; font-size: 0;'
    });

    curlyQuoteDecorationType = vscode.window.createTextEditorDecorationType({
        letterSpacing: '-1em',
        opacity: '0',
        textDecoration: 'none; font-size: 0;'
    });

    readTimeDecorationType = vscode.window.createTextEditorDecorationType({});
    lineNumberDecorationType = vscode.window.createTextEditorDecorationType({});

    statusBarItem = vscode.window.createStatusBarItem(vscode.StatusBarAlignment.Right, 100);
    statusBarItem.tooltip = 'Jav: Read time';

    dialogColor = getColorForScope('dialog') || '#ce9178';

    context.subscriptions.push(
        hiddenDecorationType,
        curlyQuoteDecorationType,
        readTimeDecorationType,
        lineNumberDecorationType,
        statusBarItem
    );

    // Selection change: only re-filter cached decorations for cursor, no rebuild
    context.subscriptions.push(
        vscode.window.onDidChangeTextEditorSelection(e => {
            const cursorLine = e.textEditor.selection.active.line;
            const cursorChar = e.textEditor.selection.active.character;

            // Only reapply if cursor actually moved
            if (cursorLine !== prevCursorLine || cursorChar !== prevCursorChar) {
                prevCursorLine = cursorLine;
                prevCursorChar = cursorChar;
                applyDecorations(e.textEditor);
                updateStatusBar(e.textEditor);
            }
        })
    );

    // Visible range change: reapply (cached data, cheap)
    context.subscriptions.push(
        vscode.window.onDidChangeTextEditorVisibleRanges(e => {
            applyDecorations(e.textEditor);
        })
    );

    // Text change: invalidate cache, rebuild
    context.subscriptions.push(
        vscode.workspace.onDidChangeTextDocument(e => {
            if (isApplyingEdits) return;

            const editor = vscode.window.activeTextEditor;
            if (editor && e.document === editor.document) {
                invalidateCache();
                applyDecorations(editor);
                updateStatusBar(editor);

                const config = getConfig();
                clearTimeout(transformTimeout);
                transformTimeout = setTimeout(() => {
                    applyTransforms(editor);
                }, config.uppercaseDebounce);
            }
        })
    );

    context.subscriptions.push(
        vscode.window.onDidChangeActiveTextEditor(editor => {
            if (editor) {
                invalidateCache();
                applyDecorations(editor);
                updateStatusBar(editor);
                applyTransforms(editor);
            } else {
                statusBarItem.hide();
            }
        })
    );

    context.subscriptions.push(
        vscode.workspace.onDidChangeConfiguration(e => {
            if (e.affectsConfiguration('jav')) {
                invalidateCache();
                const editor = vscode.window.activeTextEditor;
                if (editor) {
                    applyDecorations(editor);
                    updateStatusBar(editor);
                    applyTransforms(editor);
                }
            }
            if (e.affectsConfiguration('workbench.colorTheme') ||
                e.affectsConfiguration('editor.tokenColorCustomizations')) {
                dialogColor = getColorForScope('dialog') || '#ce9178';
                invalidateCache();
                const editor = vscode.window.activeTextEditor;
                if (editor) applyDecorations(editor);
            }
        })
    );

    // Commands
    context.subscriptions.push(
        vscode.commands.registerCommand('jav.format', () => {
            const editor = vscode.window.activeTextEditor;
            if (!editor) return;

            const document = editor.document;
            if (document.languageId !== 'jav') {
                vscode.window.showInformationMessage('Jav format only works with .jav files');
                return;
            }

            const selection = editor.selection;
            const text = selection.isEmpty
                ? document.getText()
                : document.getText(selection);

            const formatted = formatJav(text);

            editor.edit(editBuilder => {
                if (selection.isEmpty) {
                    const fullRange = new vscode.Range(
                        document.positionAt(0),
                        document.positionAt(document.getText().length)
                    );
                    editBuilder.replace(fullRange, formatted);
                } else {
                    editBuilder.replace(selection, formatted);
                }
            });
        })
    );

    context.subscriptions.push(
        vscode.commands.registerCommand('jav.toggle', async () => {
            enabled = !enabled;
            const editor = vscode.window.activeTextEditor;
            if (editor) {
                if (!enabled) {
                    await stripAllIndents(editor);
                }
                applyDecorations(editor);
                updateStatusBar(editor);
                if (enabled) {
                    await applyTransforms(editor);
                }
            }
        })
    );

    context.subscriptions.push(
        vscode.commands.registerCommand('jav.convertMd', () => {
            const editor = vscode.window.activeTextEditor;
            if (!editor) return;

            const document = editor.document;
            const selection = editor.selection;
            const text = selection.isEmpty
                ? document.getText()
                : document.getText(selection);

            const converted = mdToJav(text);
            console.log('mdToJav input length:', text.length, 'output length:', converted.length);
            console.log('changed:', text !== converted);

            editor.edit(editBuilder => {
                if (selection.isEmpty) {
                    const fullRange = new vscode.Range(
                        document.positionAt(0),
                        document.positionAt(document.getText().length)
                    );
                    editBuilder.replace(fullRange, converted);
                } else {
                    editBuilder.replace(selection, converted);
                }
            });
        })
    );

    context.subscriptions.push(
        vscode.commands.registerCommand('jav.splitToFiles', async () => {
            const editor = vscode.window.activeTextEditor;
            if (!editor) return;

            const document = editor.document;
            if (document.languageId !== 'jav') {
                vscode.window.showInformationMessage('Split only works with .jav files');
                return;
            }

            const text = document.getText();
            const lines = text.split('\n').map(l => l.replace(/\r$/, ''));

            // Find all level-1 headers (no leading tabs)
            const sections = [];
            for (let i = 0; i < lines.length; i++) {
                const match = lines[i].match(/^([^\t\n].+?)\s*(?:\{[^}]*\})?\s*\[$/);
                if (match) {
                    const title = match[1].trim();
                    const startLine = i;
                    // Walk forward to find matching close bracket
                    let depth = 1;
                    let endLine = i;
                    for (let j = i + 1; j < lines.length; j++) {
                        for (const ch of lines[j]) {
                            if (ch === '[') depth++;
                            if (ch === ']') depth--;
                            if (depth === 0) { endLine = j; break; }
                        }
                        if (depth === 0) break;
                    }
                    sections.push({
                        title,
                        content: lines.slice(startLine, endLine + 1).join('\n')
                    });
                }
            }

            if (sections.length === 0) {
                vscode.window.showInformationMessage('No level-1 headers found');
                return;
            }

            // Prompt for output folder
            const folderUri = await vscode.window.showOpenDialog({
                canSelectFolders: true,
                canSelectFiles: false,
                canSelectMany: false,
                openLabel: 'Select output folder'
            });

            if (!folderUri || folderUri.length === 0) return;

            const outputDir = folderUri[0].fsPath;
            let written = 0;

            for (const section of sections) {
                // Sanitize filename
                const safeName = section.title
                    .replace(/[<>:"/\\|?*]/g, '_')
                    .replace(/\s+/g, ' ')
                    .trim();
                const filePath = path.join(outputDir, safeName + '.jav');
                try {
                    fs.writeFileSync(filePath, section.content, 'utf8');
                    written++;
                } catch (err) {
                    vscode.window.showErrorMessage(`Failed to write ${safeName}.jav: ${err.message}`);
                }
            }

            vscode.window.showInformationMessage(`Split ${written} section(s) to ${outputDir}`);
        })
    );

    // Providers
    context.subscriptions.push(
        vscode.languages.registerDocumentSymbolProvider({ language: 'jav' }, symbolProvider),
        vscode.languages.registerFoldingRangeProvider({ language: 'jav' }, foldingProvider)
    );

    // Initial update
    if (vscode.window.activeTextEditor) {
        applyDecorations(vscode.window.activeTextEditor);
        updateStatusBar(vscode.window.activeTextEditor);
        applyTransforms(vscode.window.activeTextEditor);
    }
}

function deactivate() {
    if (transformTimeout) clearTimeout(transformTimeout);
    if (statusBarItem) statusBarItem.dispose();
}

module.exports = { activate, deactivate };

==== mdToJav.js ====

/**
 * Converts markdown-style # headers into .jav bracketed structure.
 * Only touches header lines. All other content passes through unchanged.
 */
function mdToJav(text) {
    const lines = text.split('\n').map(l => l.replace(/\r$/, ''));
    const result = [];
    const openLevels = []; // stack of heading depths currently open

    for (let i = 0; i < lines.length; i++) {
        const line = lines[i];
        const headerMatch = line.match(/^(#{1,6})\s+(.+)$/);

        if (headerMatch) {
            const level = headerMatch[1].length; // 1-6
            const title = headerMatch[2].trim();

            // Close any sections at same or deeper level
            while (openLevels.length > 0 && openLevels[openLevels.length - 1] >= level) {
                openLevels.pop();
                const indent = '\t'.repeat(openLevels.length);
                result.push(indent + ']');
            }

            // Blank line before header (unless start of file)
            if (result.length > 0 && result[result.length - 1] !== '') {
                result.push('');
            }

            const indent = '\t'.repeat(openLevels.length);
            result.push(indent + title + ' [');
            openLevels.push(level);
        } else {
            // Non-header line: indent to current depth, pass through as-is
            if (openLevels.length > 0) {
                const indent = '\t'.repeat(openLevels.length);
                // Preserve blank lines without adding trailing whitespace
                if (line.trim() === '') {
                    result.push('');
                } else {
                    result.push(indent + line.trim());
                }
            } else {
                result.push(line);
            }
        }
    }

    // Close all remaining open sections
    while (openLevels.length > 0) {
        openLevels.pop();
        const indent = '\t'.repeat(openLevels.length);
        result.push(indent + ']');
    }

    return result.join('\n');
}

module.exports = { mdToJav };
